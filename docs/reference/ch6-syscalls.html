    • 0x00 — SISTEMA (Core)
      • EVT_RESET(flags:FLG)
      • EVT_TICK_1MS(tick:FLG)
    • 0x04 — DEEP SLEEP (soppiantano i WAKE_DEEP core, ora qui)
      • EVT_WAKE_DEEP_TIMER(flags:FLG)
      • EVT_WAKE_DEEP_GPIO(flags:FLG)
      • EVT_WAKE_DEEP_TOUCH(flags:FLG)
    • 0x20 — TIMER HARDWARE
      • EVT_TIMER0_EXP(flags:FLG)
      • EVT_TIMER1_EXP(flags:FLG)
      • EVT_TIMER2_EXP(flags:FLG)
      • EVT_TIMER3_EXP(flags:FLG)
      • EVT_WATCHDOG(flags:FLG)
    • 0x40 — I/O DIGITALE & ANALOGICO
      • EVT_GPIO_BANK0(flags:FLG)
      • EVT_GPIO_BANK1(flags:FLG)
      • EVT_ADC_READY(buf:PTR,len:U16)
      • EVT_ADC_THR_CROSS(flags:FLG)
      • EVT_DAC_EMPTY
      • EVT_PWM_FAULT(flags:FLG)
    • 0x80 — COMUNICAZIONE SERIALE
      • EVT_UART0_RX(buf:PTR,len:U16)
      • EVT_UART0_TX_EMPTY
      • EVT_UART0_ERR(flags:FLG)
      • EVT_UART1_RX(buf:PTR,len:U16)
      • EVT_UART1_TX_EMPTY
      • EVT_SPI0_TRANS(buf:PTR,len:U16)
      • EVT_SPI0_TX_EMPTY
      • EVT_I2C0_SLAVE_RX(buf:PTR,len:U16)
      • EVT_I2C0_SLAVE_TX
      • EVT_I2C0_MASTER_DONE(flags:FLG)
    • 0xC0 — NETWORK & PROTOCOLLI
      • EVT_NET_LINK(flags:FLG)
      • EVT_NET_PKT_RX(buf:PTR,len:U16)
      • EVT_NET_PKT_TX_DONE
      • EVT_BT_STACK(flags:FLG)
    • 0xE0-0xF8 — SPAZIO UTENTE (portabile, Role = U)
      • EVT_USER_CUSTOM_0
      • EVT_USER_CUSTOM_1(flags:FLG)
      • EVT_USER_CUSTOM_2(value:U32)
      • EVT_USER_CUSTOM_3(buf:PTR,len:U16)
      • EVT_USER_CUSTOM_4(cmd:U32,ptr:PTR)
      • EVT_USER_CUSTOM_5(dst:PTR,src:PTR,size:U32)
      • EVT_USER_CUSTOM_6(flags:FLG,buf:PTR,len:U16)
    • Uncategorized
      • EVT_SDL_KEYDOWN(scancode:U32)
      • EVT_SDL_KEYUP(scancode:U32)
      • EVT_SDL_QUIT

      • EVT_SDL_MOUSEMOTION(x:U32,y:U32)

api 6. Syscall

Chiamate di sistema.

0x00

SISTEMA

TargetIDNomeABIDescrizione
all0x00 sys_reset ()->(ok:U32) Reset VM
all0x01 sys_version ()->(version:U32) Versione VM
all0x02 sys_features ()->(flags:FLG) Feature abilitate
all0x03 sys_millis ()->(ms:U32) Tempo corrente in ms
all0x04 sys_static_ptr (addr:U32)->(ptr:PTR) Converte indirizzo .STATIC in puntatore RAM scrivibile
all0x05 sys_event_id ()->(id:U32) ID Evento corrente (0xFFFFFFFF = nessuno)

0x10

HAL

TargetIDNomeABIDescrizione
all0x10 sys_print (fmt:PTR,args:PTR)->(ok:U32) Stampa stringa formattata
all0x13 sys_print_status () Stampa stato VM

0x20

GPIO

TargetIDNomeABIDescrizione
esp32, esp32s30x20 sys_gpio_set_dir (pin:U32,dir:U32)->(ok:U32) GPIO set direction
esp32, esp32s30x21 sys_gpio_write (pin:U32,value:U32)->(ok:U32) GPIO write
esp32, esp32s30x22 sys_gpio_read (pin:U32)->(value:U32) GPIO read

0x30

UART

TargetIDNomeABIDescrizione
esp32, esp32s30x30 sys_uart_send (byte:U32)->(ok:U32) UART send
esp32, esp32s30x31 sys_uart_read (port:U32)->(byte:U32) UART read

0x40

Timer

TargetIDNomeABIDescrizione
esp32, esp32s30x40 sys_delay (ms:U32)->(ok:U32) Delay ms

0xD0

SDL2 GRAPHICS

TargetIDNomeABIDescrizione
linux0xD0 sys_gfx_open (width:U32,height:U32,title:PTR)->(handle:HND) Apre finestra
linux0xD1 sys_gfx_flush (handle:HND,buf:PTR,size:U32)->(ok:U32) Flush buffer pixel
linux0xD2 sys_gfx_poll (handle:HND)->(event:U32) Legge evento
linux0xD3 sys_gfx_close (handle:HND)->(ok:U32) Chiude finestra
linux0xD4 sys_gfx_info (handle:HND,buf:PTR)->(ok:U32) Legge info schermo
linux0xD5 sys_gfx_get_input_addr (handle:HND)->(addr:PTR) Indirizzo input word
linux0xD6 sys_gfx_read_input (handle:HND)->(scancode:U32) Legge scancode
linux0xD7 sys_gfx_get_mouse (handle:HND)->(x:U32,y:U32,btn:FLG) Stato mouse
linux0xD8 sys_gfx_get_winsize (handle:HND)->(width:U32,height:U32) Dimensione finestra
linux0xD9 sys_text_print (handle:HND,str:PTR,x:U32,y:U32,fg:U32,bg:U32,fb:PTR)->(ok:U32) Stampa testo bitmap
